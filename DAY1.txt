DEFINE class UndoRevert
{
ATTRIBUTE text
ATTRIBUTE originalText
ATTRIBUTE undoStack (stack of strings)

CONSTRUCTOR(text,originalText)
{
SET this.text=text
SET originalText=text
INITIALIZE undoStack AS empty stack 
}

METHOD addText(newText)
{
PUSH text INTO undoStack
ASSIGN text + newText INTO text
}

METHOD Undo()
{
IF undoStack IS NOT empty THEN
ASSIGN text TO POP FROM undoStack
}

METHOD revert()
{
SET text TO originalText
CLEAR undoStack
}
}
END
