 	ATTRIBUTES: RED,GREEN,YELLOW

 	ATTRIBUTE currentLight

 	END CONSTRUCTOR
 	ATTRIBUTE durationForRed

	ASSIGN durationForRed=5000

 	ASSIGN durationForYellow=4000

 	METHOD switchLight()
 		ATTRIBUTE now
 		ASSIGN now=currentTime
 
 		ATTRIBUTE timePassed=now-lastChangeTime
 
 		IF currentLight IS RED AND timePassed IS Greater equal to  durationForRed THEN
 			ASSIGN currentLight=GREEN
 			ASSIGN lastChangeTime=currentTime
 		END IF
 		IF currentLight IS GREEN AND timePassed IS Greater equal to durationForGREEN THEN
 			ASSIGN currentLight=YELLOW
 			ASSIGN lastChangeTime=currentTime
 		END IF
 		IF currentLight IS YELLOW AND timePassed IS Greater equal to durationForYellw THEN
 			ASSIGN currentLight=RED
 			ASSIGN lastChangeTime=currentTime
 		END IF
 	END METHOD
 	METHOD getCurrentLight()
 		PRINT currentLight
 	END METHOD
